## 二、任务 1：补前端错误码中文提示

修改文件：

```text
frontend/src/pages/DirectorStudioPage.tsx
frontend/src/pages/directorStudioLogic.ts
```

新增统一错误转换函数：

```ts
function normalizeDirectorErrorMessage(error: unknown): string {
  const raw =
    typeof error === 'string'
      ? error
      : error instanceof Error
        ? error.message
        : JSON.stringify(error ?? '')

  if (raw.includes('CRITICAL_ARTIFACT_SYNC_FAILED')) {
    return '关键产物写入失败，最终视频无法进入项目产物库。请重新执行当前步骤。'
  }

  if (raw.includes('ARTIFACT_MANIFEST_INVALID')) {
    return '本地任务返回的产物信息不完整，无法写入项目产物库。请重新执行当前步骤。'
  }

  if (raw.includes('RENDER_DEPENDENCY_MISSING')) {
    return '当前项目尚不满足最终渲染条件，请确认预览、本地执行器和前置产物状态。'
  }

  if (raw.includes('PACKAGE_DEPENDENCY_MISSING')) {
    return '最终视频尚未通过质量检查，暂时不能打包。'
  }

  return raw || '操作失败，请稍后重试。'
}
```

在所有 `catch` 中使用：

```ts
setError(normalizeDirectorErrorMessage(error))
```

如果当前页面有 `rawError` 或高级详情区域，保留原始错误：

```ts
setRawError(
  typeof error === 'string'
    ? error
    : error instanceof Error
      ? error.message
      : JSON.stringify(error ?? '')
)
```

用户默认看到中文解释，不要只显示后端错误码。

## 三、任务 2：确认导出页只依赖 VIDEO artifact

继续保持当前规则：

```ts
const video = artifacts.find((artifact) => artifact.kind === 'VIDEO')
const videoReady = video?.status === 'valid' && Boolean(video.storageRef)
```

不要改成根据默认路径推断：

```text
local://projects/{projectId}/renders/final.mp4
```

判断最终视频是否存在，只能依据：

```text
VIDEO artifact
status = valid
storageRef 非空
```

如果没有 VIDEO artifact，展示：

```text
最终视频尚未生成
```

```go
video, err := artifactRepo.FindCurrentByStageAndKind(ctx, projectID, "render", "VIDEO")

require.NoError(t, err)
require.NotNil(t, video)
require.Equal(t, "final-video", video.UnitID)
```
